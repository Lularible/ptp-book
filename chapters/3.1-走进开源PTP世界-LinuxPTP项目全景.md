## 项目简介

LinuxPTP是Richard Cochran开发的**IEEE 1588 PTP协议开源实现**，专为Linux系统设计。

### 核心特点

```text
特性一：原生Linux支持
- 使用Linux内核最新的时间戳API
- 支持PTP硬件时钟（PHC）子系统
- 利用SO_TIMESTAMPING套接字选项

特性二：完整协议实现
- 支持普通时钟（OC）
- 支持边界时钟（BC）
- 支持透明时钟（TC，包括E2E和P2P）

特性三：多种传输方式
- UDP/IPv4
- UDP/IPv6
- 原始以太网（IEEE 802.3）

特性四：多种Profile支持
- 默认1588 Profile
- 电信Profile（G.8265.1、G.8275.1、G.8275.2）
- 企业Profile
- 汽车Profile

特性五：高级功能
- 单播操作
- 安全认证（AUTHENTICATION TLV）
- NetSync Monitor协议
- IEEE 802.1AS支持（端站角色）
```

### 系统要求

```text
内核要求：Linux 3.0或更新版本

检查网卡是否支持PTP硬件时间戳：
$ ethtool -T eth0

期望输出包含：
  hardware-transmit     (SOF_TIMESTAMPING_TX_HARDWARE)
  hardware-receive      (SOF_TIMESTAMPING_RX_HARDWARE)
  hardware-raw-clock    (SOF_TIMESTAMPING_RAW_HARDWARE)
  PTP Hardware Clock: 1
```

## 核心数据结构

### 时钟类型枚举

```c
/* clock.h:38-44 （逐字摘录，中文注释为本书所加） */

enum clock_type {
    CLOCK_TYPE_ORDINARY   = 0x8000,  /* 普通时钟 */
    CLOCK_TYPE_BOUNDARY   = 0x4000,  /* 边界时钟 */
    CLOCK_TYPE_P2P        = 0x2000,  /* P2P透明时钟 */
    CLOCK_TYPE_E2E        = 0x1000,  /* E2E透明时钟 */
    CLOCK_TYPE_MANAGEMENT = 0x0800,  /* 管理节点 */
};
```

**设计解读**：

> 注：这里要更正一个容易误读的地方。这五个常量的值确实都是互不相同的 2 的幂（0x8000、0x4000、0x2000、0x1000、0x0800），看上去像位掩码，但 **linuxptp 不按位测试它们**。`clock_type` 是一个普通枚举，变量 `type` 在任一时刻只代表一种类型，代码里一律用 `switch` 或相等比较：

```c
/* ptp4l.c:214-220 —— 读配置后做类型校验 */
type = config_get_int(cfg, NULL, "clock_type");
switch (type) {
case CLOCK_TYPE_ORDINARY:
	if (cfg->n_interfaces > 1) {
		type = CLOCK_TYPE_BOUNDARY;
	}
	break;

/* clock.c:1103-1112 —— 管理节点类型直接拒绝 */
switch (type) {
case CLOCK_TYPE_ORDINARY:
case CLOCK_TYPE_BOUNDARY:
case CLOCK_TYPE_P2P:
case CLOCK_TYPE_E2E:
	c->type = type;
	break;
case CLOCK_TYPE_MANAGEMENT:
	return NULL;
}

/* port.c:3553-3569 —— 按类型挑选 dispatch/event 函数对（节选） */
switch (type) {
case CLOCK_TYPE_ORDINARY:
case CLOCK_TYPE_BOUNDARY:
	p->dispatch = bc_dispatch;
	p->event = bc_event;
	break;
case CLOCK_TYPE_P2P:
	p->dispatch = p2p_dispatch;
	p->event = p2p_event;
	break;
/* 以下为 e2e_dispatch / err_log_name 两个分支，见源码 */
}

/* port.c:3674-3678、3679-3683 —— 相等比较，不是按位测试 */
if (!port_is_uds(p) && type == CLOCK_TYPE_P2P &&
    p->delayMechanism != DM_P2P) {
	pr_err("%s: P2P TC needs P2P ports", p->log_name);
	goto err_uc_service;
}
if (!port_is_uds(p) && type == CLOCK_TYPE_E2E &&
    p->delayMechanism != DM_E2E) {
	pr_err("%s: E2E TC needs E2E ports", p->log_name);
	goto err_uc_service;
}
```

选 2 的幂只是为了让每个枚举值有独立的 16 位模式（`clock_type` 要按 `UInteger16` 写进 `CLOCK_DESCRIPTION` 管理属性：`tlv.h:458` 声明 `UInteger16 *clockType;`，`port.c:950` 赋值 `*cd->clockType = clock_type(target->clock);`），并不代表可以组合。

### 端口状态枚举

```c
/* fsm.h:24-35 （逐字摘录，中文注释为本书所加） */

enum port_state {
    PS_INITIALIZING = 1,    /* 初始化 */
    PS_FAULTY,              /* 故障 */
    PS_DISABLED,            /* 禁用 */
    PS_LISTENING,           /* 监听 */
    PS_PRE_MASTER,          /* 预备主 */
    PS_MASTER,              /* 主时钟 */
    PS_PASSIVE,             /* 被动 */
    PS_UNCALIBRATED,        /* 未校准 */
    PS_SLAVE,               /* 从时钟 */
    PS_GRAND_MASTER,        /* 主时钟（非标准扩展）*/
};
```

**与IEEE 1588的对应关系**：

```text
IEEE 1588 定义的 9 种状态，linuxptp 一一对应（fsm.h:25-33）：
1. INITIALIZING      → PS_INITIALIZING
2. FAULTY            → PS_FAULTY
3. DISABLED          → PS_DISABLED
4. LISTENING         → PS_LISTENING
5. PRE_MASTER        → PS_PRE_MASTER
6. MASTER            → PS_MASTER
7. PASSIVE           → PS_PASSIVE
8. UNCALIBRATED      → PS_UNCALIBRATED
9. SLAVE             → PS_SLAVE

linuxptp 的第 10 个状态（非标准扩展）：
PS_GRAND_MASTER（fsm.h:34，源码注释原文就是 /*non-standard extension*/）
含义：本时钟就是整个时钟域的根，而不只是某条路径上的 MASTER 端口。
```

> 注：本节只列出这份枚举，逐状态的转移规则与两套状态机（`ptp_fsm` / `ptp_slave_fsm`）的完整解析见 **3.3**；3.3 不再重复上面这张标准对应表。

### 伺服状态枚举

```c
/* servo.h:44-67（成员名与顺序逐字一致；源码在四个成员之间
   还各有一段 doxygen 注释，此处从略） */

enum servo_state {
    SERVO_UNLOCKED,     /* 未锁定：需要更多数据 */
    SERVO_JUMP,         /* 跳变：需要大步调整 */
    SERVO_LOCKED,       /* 锁定：正在跟踪 */
    SERVO_LOCKED_STABLE,/* 稳定锁定：偏差在阈值内 */
```

**状态转换逻辑**：

```text
状态转换流程（以下为改写版整理，真实源码见所注行号）：

先记住一件事：状态由两层共同给出，不是某一个 switch ——
  ① 具体伺服的 sample() 回调先给出本轮状态（默认是 PI，pi.c:64-155）；
  ② 外层 servo_sample() 再做计数与“升级”（servo.c:116-150）。

SERVO_UNLOCKED
      │
      │  PI 伺服 count=0：只存下第一个样本，仍报 UNLOCKED（pi.c:75-80）
      │  PI 伺服 count=1：还要过两道前置检查才肯估频率 ——
      │    · 第一个样本必须早于第二个，否则 count 退回 0（pi.c:86-90）；
      │    · 两点间隔必须 >= 0.016/ki，上限 1000.0 秒（pi.c:92-102）。
      ▼
   估算频率偏差 drift，并按 servo->max_frequency 钳位（pi.c:104-111）
      │
      ├── 超阈值：首次更新且超 first_step_threshold，
      │           或超 step_threshold               → SERVO_JUMP（pi.c:113-118）
      │           跳变的同时 curr_offset_values 重置、
      │           first_update 清 0（servo.c:130-133）
      │
      └── 未超阈值                                   → SERVO_LOCKED（pi.c:119-120）
                                                          （不是必经 JUMP！）

SERVO_LOCKED
      │
```

### clock_poll函数

```c
/* clock_poll()。以下为改写版（节选），真实源码见 clock.c:1797-1886 */

int clock_poll(struct clock *c)
{
    clock_check_pollfd(c);                     /* clock.c:1805 */

    /* 步骤1：poll 等待事件。
       没有 c->n_pollfd 这个字段——个数是现算的：
       nports+2 是“每个网络端口 + 两个 UDS 端口”，
       N_CLOCK_PFD = N_POLLFD + 1（clock.c:50），
       多加的那个 1 是每端口的故障定时器 fd */
    cnt = poll(c->pollfd, (c->nports + 2) * N_CLOCK_PFD, -1);   /* :1806 */
    if (cnt < 0) {
        if (EINTR == errno) {
            return 0;                          /* :1808-1809 */
        } else {
            pr_emerg("poll failed");
            return -1;
        }
    } else if (!cnt) {
        return 0;
    }

    cur = c->pollfd;                           /* :1818 */

    /* 步骤2：遍历端口。
       也不是“拿 fd 反查端口”——端口链表本身就是有序的，
       每端口固定占 N_CLOCK_PFD 个 pollfd 槽位，所以一边遍历链表
       一边把 cur 往后推进一格即可 */
    LIST_FOREACH(p, &c->ports, list) {         /* clock.c:1820 */
        for (i = 0; i < N_POLLFD; i++) {       /* :1822 */
            if (cur[i].revents & (POLLIN|POLLPRI|POLLERR)) {
                prior_state = port_state(p);
                if (cur[i].revents & POLLERR) {
                    /* 真实代码是 sk_get_error(cur[i].fd) + pr_err 打印
                       出错描述，见 clock.c:1826-1829 */
                    event = EV_FAULT_DETECTED; /* :1825-1830 */
                } else {
                    event = port_event(p, i);  /* :1832 */
                }

                /* 关键：这两类事件只是把 SDE 标志立起来，
                   并不在这里做 BMCA，见下面步骤4 */
                if (EV_STATE_DECISION_EVENT == event) {
                    c->sde = 1;                /* :1834-1836 */
                }
                if (EV_ANNOUNCE_RECEIPT_TIMEOUT_EXPIRES == event) {
                    c->sde = 1;                /* :1837-1839 */
                }

                /* 无条件把事件交给端口状态机（第三个参数是 0，不是 mdiff） */
                port_dispatch(p, event, 0);    /* :1840 */

                if ((PS_FAULTY == port_state(p)) &&
                    (prior_state != PS_FAULTY)) {
                    clock_fault_timeout(p, 1);  /* :1842-1845 */
                    break;
                }
            }
        }
        /* 每端口第 N_POLLFD 个槽位是故障定时器：到期即清除故障，
           且仅在链路 up 时才派 EV_FAULT_CLEARED（clock.c:1849-1858） */
        cur += N_CLOCK_PFD;                    /* :1860 */
    }

    /* 步骤3：两个 UDS 管理端口（clock.c:1863-1878） */
    for (i = 0; i < N_POLLFD; i++) {
        if (cur[i].revents & (POLLIN|POLLPRI)) {
            event = port_event(c->uds_rw_port, i);
            if (EV_STATE_DECISION_EVENT == event) {
                c->sde = 1;                    /* :1867-1869 */
            }
        }
    }
    cur += N_CLOCK_PFD;
    /* uds_ro_port 只读，同样轮询一轮（:1873-1877） */

    /* 步骤4：真正的 BMCA 在这里，被推迟到整轮 poll 处理完之后 */
    if (c->sde) {                              /* clock.c:1880-1883 */
        handle_state_decision_event(c);        /* 定义在 clock.c:2207 */
        c->sde = 0;
    }

    clock_prune_subscriptions(c);              /* clock.c:1884 */
    return 0;
}
```

### 配置解析流程

```c

int config_read(const char *name, struct config *cfg)   /* 形参叫 name，不是 path */
{
    enum config_section current_section = UNKNOWN_SECTION;  /* config.c:821-827 */
    enum parser_result parser_res;
    FILE *fp;
    char buf[1024], *line, *c;            /* config.c:824 —— 没有 MAX_LINE 这个宏 */

    struct interface *current_port = NULL;

    /* "-" 开头就读标准输入，否则才 fopen */
    fp = 0 == strncmp(name, "-", 2) ? stdin : fopen(name, "r");   /* :829 */
    if (!fp) {
        fprintf(stderr, "failed to open configuration file %s: %m\n", name);
        return -1;                        /* :831-834 */
    }

    for (line_num = 1; fgets(buf, sizeof(buf), fp); line_num++) { /* :836 */
        c = buf;
        while (isspace(*c))               /* 先去前导空白 */
            c++;
        if (*c == '#' || *c == '\n' || *c == '\0')
            continue;                     /* 跳过注释和空行，config.c:844-845 */
        line = c;
        /* …去掉行尾空白与 \n（config.c:849-852）… */

        /* 分三种行处理，各自一个独立的解析函数： */

        /* 1) [global] / [eth0] 这样的 section 头 */

        if (parse_setting_line(line, &option, &value))
            goto parse_error;             /* :881-886 */
        check_deprecated_options(&option);            /* :888 */
        parser_res = parse_item(cfg, 0,
                     current_section == GLOBAL_SECTION ?
                     NULL : interface_name(current_port),
                     option, value);      /* :890-892 */
        switch (parser_res) {             /* :893-914 */
        /* 五个取值定义在 util.h:261-267 */
        case NOT_PARSED: case BAD_VALUE: case MALFORMED: case OUT_OF_RANGE:
            goto parse_error;             /* 各自 fprintf 说明，见源码 */
        case PARSED_OK:
            break;
        }
    }

    fclose(fp);
    return 0;                             /* config.c:917-918 */

parse_error:
    fprintf(stderr, "failed to parse configuration file %s\n", name);
    fclose(fp);
    return -2;                            /* config.c:920-923 */
}
```

## 构建和安装

### 编译

```bash
# 进入源码目录
cd linuxptp

# 编译（默认使用系统内核头文件）
make

# 如果使用自定义内核
make KBUILD_OUTPUT=/path/to/kernel/build

# 安装（默认安装到/usr/local）
make install

# 指定安装路径
make prefix=/opt/ptp install
```

### 运行

```bash
# 不指定 -f 时的运行示例（没有“默认配置文件”，全靠命令行与内置默认值）
ptp4l -i eth0 -S -m

# 参数说明：
# -i eth0  : 使用eth0接口
# -S       : 使用软件时间戳
# -m       : 输出到stdout

# 使用配置文件运行。仓库样例 configs/default.cfg 只有 [global] 段、
# 没有接口段，所以 -i 仍必须单独给出，否则会报 "no interface specified"
# （ptp4l.c:208-212）
ptp4l -f configs/default.cfg -i eth0

# 运行边界时钟
ptp4l -i eth0 -i eth1 -m

# 使用硬件时间戳
ptp4l -i eth0 -H -m
```
