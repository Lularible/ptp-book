# 3.6 PHC操作与时钟调整：与硬件时钟对话

## 从用户态到内核态

LinuxPTP运行在用户态，但需要操作内核的PTP硬件时钟（PHC）。

这是如何实现的？

---

## Linux PHC子系统

### PHC是什么

```text
PTP Hardware Clock（PHC）是Linux内核提供的子系统：
- 抽象各种硬件时钟设备
- 提供统一的用户态接口
- 支持多种操作：读取、设置、调整

设备文件：
/dev/ptp0   # 第一个PHC设备
/dev/ptp1   # 第二个PHC设备
...

查看系统中的PHC：
$ ls -l /dev/ptp*
crw-rw---- 1 root root 248, 0 Apr  1 10:00 /dev/ptp0
```

### PHC的内核接口

```c
/* Linux内核头文件 <linux/ptp_clock.h> */

struct ptp_clock_caps {
    int max_adj;        /* 最大频率调整（ppb） */
    int n_alarm;        /* 告警数量 */
    int n_ext_ts;       /* 外部时间戳数量 */
    int n_per_out;      /* 周期性输出数量 */
    int pps;            /* 是否支持PPS */
    int n_pins;         /* 引脚数量 */
    int cross_timestamping; /* 是否支持交叉时间戳 */
    int adjust_phase;   /* 是否支持相位调整 */

    int rsv[11];        /* 保留给将来扩展 */
};
```

## PHC操作函数

### phc_open：打开PHC设备

```c
/* phc.c, 第42-67行 */

clockid_t phc_open(const char *phc)
{
    clockid_t clkid;
    struct timespec ts;
    struct timex tx;
    int fd;

    memset(&tx, 0, sizeof(tx));

    /* 步骤1：打开设备文件 */
    fd = open(phc, O_RDWR);
    if (fd < 0)
        return CLOCK_INVALID;

    /* 步骤2：将文件描述符转换为clockid_t */
    clkid = FD_TO_CLOCKID(fd);
    
    /* 步骤3：验证clockid是否有效 */
    if (clock_gettime(clkid, &ts)) {
        close(fd);
        return CLOCK_INVALID;
    }
    
    /* 步骤4：验证是否可以调整时钟 */
    if (clock_adjtime(clkid, &tx)) {
        close(fd);
        return CLOCK_INVALID;
    }

    return clkid;
}
```

**FD_TO_CLOCKID宏**：

```c
/* missing.h, 第49-51行 */

#define CLOCKFD 3
#define FD_TO_CLOCKID(fd)	((clockid_t) ((((unsigned int) ~fd) << 3) | CLOCKFD))
#define CLOCKID_TO_FD(clk)	((unsigned int) ~((clk) >> 3))
```

**工作原理**：

```text
Linux时钟API使用clockid_t标识时钟：
- CLOCK_REALTIME：系统实时时钟
- CLOCK_MONOTONIC：单调递增时钟
- CLOCKFD：动态时钟（通过文件描述符）

PTP设备文件描述符 → clockid_t转换：
1. 打开/dev/ptp0 → 获得fd（如fd=10）
2. FD_TO_CLOCKID(10) → clockid_t
3. 用这个clockid调用clock_gettime等函数

这允许用户态程序像操作普通时钟一样操作PHC！
```

### phc_close：关闭PHC设备

```c
/* phc.c, 第69-75行 */

void phc_close(clockid_t clkid)
{
    if (clkid == CLOCK_INVALID)
        return;

    close(CLOCKID_TO_FD(clkid));
```

### phc_max_adj：获取最大频率调整

```c
/* phc.c, 第87-101行 */

int phc_max_adj(clockid_t clkid)
{
    int max;
    struct ptp_clock_caps caps;

    /* 通过ioctl获取能力 */
    if (phc_get_caps(clkid, &caps))
        return 0;

    max = caps.max_adj;

    /* 32位平台的特殊处理 */
    if (BITS_PER_LONG == 32 && max > MAX_PPB_32)
        max = MAX_PPB_32;

    return max;
}

static int phc_get_caps(clockid_t clkid, struct ptp_clock_caps *caps)
{
    int fd = CLOCKID_TO_FD(clkid), err;

    err = ioctl(fd, PTP_CLOCK_GETCAPS, caps);
    if (err)
        perror("PTP_CLOCK_GETCAPS");
    return err;
}
```

**32位平台限制**：

```c
/* phc.c, 第37-38行 */

#define BITS_PER_LONG   (sizeof(long)*8)
#define MAX_PPB_32      32767999  /* (2^31 - 1) / 65.536 */
```

```text
为什么需要这个限制？

struct timex {
    long freq;    /* 频率偏移：2^16缩放的ppm，即 ppb × 65.536 */
    ...
};

（换算系数65.536在clockadj.c里出现三次：设置时×65.536（:64）、
  读取时÷65.536（:81）、算最大能力时也÷65.536（:138））

32位平台：long是32位，范围±2^31
最大值：(2^31 - 1) / 65.536 ≈ 32,767,999 ppb

如果PHC硬件支持更大的调整范围，需要限制到这个值。
```

**PPS（Pulse Per Second）**：

```text
PPS是一种硬件信号：
- 每秒产生一个脉冲
- 精度可达纳秒级
- 用于精确的时间同步

用途：
- 外部时间源（GPS）输入
- 时间戳触发
- 频率校准

LinuxPTP可以利用PPS：
- 如果PHC支持PPS
- 可以从外部源获取精确时间
- 提高同步精度
```

### phc_has_writephase：检查相位调整支持

```c
/* phc.c, 第131-139行 */

int phc_has_writephase(clockid_t clkid)
{
    struct ptp_clock_caps caps;

    if (phc_get_caps(clkid, &caps)) {
        return 0;
    }
    
    return caps.adjust_phase;
}
```

**相位调整能力**：

```text
传统频率调整：
- 通过调整振荡器频率
- 相位变化是累积的
- 适合长期稳定

相位调整（硬件写入相位）：
- 直接修改时钟相位
- 不改变频率
- 立即生效，无累积误差

支持adjust_phase的硬件：
- 可以更精确地调整相位
- 实现"一步"时钟
- 适合高精度场景
```

## 时钟调整函数

### clockadj_set_freq：设置频率

```c
/* clockadj.c, 第51-70行 */

int clockadj_set_freq(clockid_t clkid, double freq)
{
    struct timex tx;
    memset(&tx, 0, sizeof(tx));

    /* 系统时钟的特殊处理 */
    if (clkid == CLOCK_REALTIME && realtime_nominal_tick) {
        tx.modes |= ADJ_TICK;
        tx.tick = round(freq / 1e3 / realtime_hz) + realtime_nominal_tick;
        freq -= 1e3 * realtime_hz * (tx.tick - realtime_nominal_tick);
    }

    /* 设置频率 */
    tx.modes |= ADJ_FREQUENCY;
    tx.freq = (long) (freq * 65.536);
    
    if (clock_adjtime(clkid, &tx) < 0) {
        pr_err("failed to adjust the clock: %m");
        return -1;
    }
    return 0;
}
```

**频率调整的单位转换**：

```text
输入：freq（ppb，十亿分之一）
内核：tx.freq（单位：ppb × 65.536）

转换公式：
tx.freq = freq × 65.536

为什么是65.536？

历史原因：NTP使用"SHIFT_HZ = 16"
freq字段是2^16缩放的ppm（百万分之一）
1 ppm = 1000 ppb
所以：tx.freq = freq_ppb × 2^16 / 1000
               = freq_ppb × 65.536

示例：
freq = 100 ppb
tx.freq = 100 × 65.536 = 6554
```

**系统时钟的特殊处理**：

```c
if (clkid == CLOCK_REALTIME && realtime_nominal_tick) {
    tx.modes |= ADJ_TICK;
    tx.tick = round(freq / 1e3 / realtime_hz) + realtime_nominal_tick;
    freq -= 1e3 * realtime_hz * (tx.tick - realtime_nominal_tick);
}
```

```text
为什么系统时钟需要特殊处理？

系统时钟（CLOCK_REALTIME）：
- 基于内核的tick
- tick长度可以调整
- 允许更大的频率调整范围

方法：
1. 调整tick长度（粗调）
2. 调整freq字段（精调）
3. 两者结合，扩大调整范围

示例：
realtime_hz = 1000（USER_HZ）
realtime_nominal_tick = 1000（微秒）

freq = 1e8 ppb（即10%的大偏差，才需要tick粗调）

粗调：
tx.tick = round(1e8 / 1e3 / 1000) + 1000 = 1100微秒
（tick从1000微秒增加到1100微秒，相当于频率加快10%）

精调：
freq -= 1e3 × 1000 × (1100 - 1000) = 1e8
（10%全部由tick承担，freq字段剩余0）

效果：
tick完成10%的粗调整
freq字段只需精调剩余部分
```

### clockadj_get_freq：获取当前频率

```c
/* clockadj.c, 第72-86行 */

double clockadj_get_freq(clockid_t clkid)
{
    double f = 0.0;
    struct timex tx;
    memset(&tx, 0, sizeof(tx));
    
    if (clock_adjtime(clkid, &tx) < 0) {
        pr_err("failed to read out the clock frequency adjustment: %m");
        exit(1);
    } else {
        f = tx.freq / 65.536;
        if (clkid == CLOCK_REALTIME && realtime_nominal_tick && tx.tick)
            f += 1e3 * realtime_hz * (tx.tick - realtime_nominal_tick);
    }
    return f;
}
```

**逆转换**：

```text
读取时的单位转换：
freq_ppb = tx.freq / 65.536

系统时钟需要加上tick调整：
freq_ppb += 1e3 × hz × (tick - nominal_tick)
```

### clockadj_set_phase：设置相位偏移

```c
/* clockadj.c, 第88-100行 */

int clockadj_set_phase(clockid_t clkid, long offset)
{
    struct timex tx;
    memset(&tx, 0, sizeof(tx));

    tx.modes = ADJ_OFFSET | ADJ_NANO;
    tx.offset = offset;
    
    if (clock_adjtime(clkid, &tx) < 0) {
        pr_err("failed to set the clock offset: %m");
        return -1;
    }
    return 0;
}
```

**ADJ_OFFSET的作用**：

```text
ADJ_OFFSET模式：
- 设置相位偏移
- 时钟逐渐调整，不是立即跳变
- 使用内核的PLL调整

参数：
tx.offset = offset（这里是纳秒）
ADJ_NANO（missing.h:37-39，值0x2000）表示offset单位按纳秒解释；
不带这个标志时，tx.offset 的单位是微秒——见下文 struct timex 的注释

工作原理：
- 时钟偏移offset纳秒
- 内核逐渐调整，消除偏移
- 调整速度由time_constant控制

适用场景：
- 小偏差（微秒级）
- 平滑调整，不影响应用
- 在ptp4l里只走 write_phase_mode 且 SERVO_LOCKED_STABLE 这一条路（clock.c:2076-2081）
```

### clockadj_step：时钟跳变

```c
/* clockadj.c, 第102-127行 */

int clockadj_step(clockid_t clkid, int64_t step)
{
    struct timex tx;
    int sign = 1;
    if (step < 0) {
        sign = -1;
        step *= -1;
    }
    memset(&tx, 0, sizeof(tx));
    
    tx.modes = ADJ_SETOFFSET | ADJ_NANO;
    tx.time.tv_sec  = sign * (step / NS_PER_SEC);
    tx.time.tv_usec = sign * (step % NS_PER_SEC);
    
    /* 确保tv_usec非负 */
    if (tx.time.tv_usec < 0) {
        tx.time.tv_sec  -= 1;
        tx.time.tv_usec += 1000000000;
    }
    
    if (clock_adjtime(clkid, &tx) < 0) {
        pr_err("failed to step clock: %m");
        return -1;
    }
    return 0;
}
```

**时钟跳变的含义**：

```text
ADJ_SETOFFSET模式：
- 立即修改时钟值
- 时间会"跳"到新值
- 可能导致应用问题

参数：
step = +1000000000：时钟向前跳1秒
step = -1000000000：时钟向后跳1秒

处理负数step：
1. step = -1500000000（-1.5秒）
2. sign = -1
3. step = 1500000000
4. tx.time.tv_sec = -1
5. tx.time.tv_usec = -500000000
6. 调整：tv_sec -= 1, tv_usec += 1000000000
   → tv_sec = -2, tv_usec = 500000000

注意：
tx.time是timeval结构：
struct timeval {
    time_t tv_sec;    /* 秒 */
    suseconds_t tv_usec; /* 微秒 */
};

虽然字段名是 tv_usec，但在 ADJ_NANO 标志下其语义就是纳秒：
代码写入的是 step 除以 NS_PER_SEC 后的纳秒余数（clockadj.c:111-113），
此时 tv_usec 的取值范围是 [0, 999999999]。
```

**set_phase vs step**：

```text
set_phase：
- 渐进调整
- 用PLL逐渐消除偏差
- 无时间跳变
- 适合小偏差

step：
- 立即跳变
- 时间直接修改
- 可能导致应用问题
- 适合大偏差

选择依据（clock_synchronize，clock.c:2007-2107；分支switch在:2052-2088）：
- 正常跟踪的小偏差：由PI控制器估计频率并调用 clockadj_set_freq 调频，不用 set_phase
- |offset| 超过 step_threshold（或首次的first_step_threshold）→ 伺服返回SERVO_JUMP，走 clockadj_step 跳变
- clockadj_set_phase 仅在 write_phase_mode 且状态为 SERVO_LOCKED_STABLE 时使用（clock.c:2076-2079）
```

### 函数原型


**这是Linux特有的系统调用**：

```text
标准POSIX：
```

## PHC与系统时钟同步

本章讲的是"怎么调一个钟"；把 PHC 的时间搬到系统时钟上是另一个程序 `phc2sys` 的活，**它有专门的一节（3.11）**，这里只留一个最小示例，不重复展开。

```bash
# 源=PHC，目标=系统时钟，并等待 ptp4l
phc2sys -s /dev/ptp0 -c CLOCK_REALTIME -w
```

三个参数说明：

```text
 -c [dev|name]  time sink device (CLOCK_REALTIME)      目标时钟（默认就是系统钟）
 -s [dev|name]  time source device                     时间源
 -w             wait for ptp4l                         等待 ptp4l
```

## 实际应用示例

### 在LinuxPTP中使用PHC

```c
/* clock.c:1081-1445，改写：只保留"选哪个时钟 + 最大调频能力从哪来 + 伺服怎么建"这条主线 */

struct clock *clock_create(enum clock_type type, struct config *config,
                           const char *phc_device)
{
    int max_adj = 0, phc_index, sw_ts;   /* clock.c:1084：max_adj是局部变量 */
    char phc[32];
    struct clock *c = &the_clock;        /* clock.c:1088：静态单例，不是calloc来的 */
    double fadj = 0.0;

    /* ……此处源码先确定 phc_index（clock.c:1206-1228）：
       free_running 或软件/legacy时间戳 → -1；
       phc_device 形如 "/dev/ptpN" → sscanf 出 N；
       配置项 phc_index >= 0 → 用它；
       否则从网卡 tsinfo 自动探测，探不到就报错返回 NULL …… */

    /* 四个互斥分支：选时钟，并各自取出"最大能调多少"（clock.c:1296-1328） */
    if (c->free_running) {                 /* ① free_running 配置项 */
        c->clkid = CLOCK_INVALID;          /* 不碰任何时钟，也不会回退 */
        if (timestamping == TS_SOFTWARE || timestamping == TS_LEGACY_HW)
            c->utc_timescale = 1;
    } else if (phc_index >= 0) {           /* ② 有明确的 /dev/ptpN */
        snprintf(phc, sizeof(phc), "/dev/ptp%d", phc_index);
        c->clkid = phc_open(phc);
        if (c->clkid == CLOCK_INVALID) {
            pr_err("Failed to open %s: %m", phc);
            return NULL;                   /* 打不开就失败，绝不悄悄换钟 */
        }
        max_adj = phc_max_adj(c->clkid);   /* ← 只有这一支用 phc_max_adj() */
        if (!max_adj) {
            pr_err("clock is not adjustable");
            return NULL;
        }
        clockadj_init(c->clkid);
    } else if (phc_device) {               /* ③ 给了设备名，但不是 /dev/ptpN 形式 */
        c->clkid = phc_open(phc_device);
        if (c->clkid == CLOCK_INVALID) {
            pr_err("Failed to open %s: %m", phc_device);
            return NULL;
        }
        max_adj = clockadj_max_freq(c->clkid);  /* ← 这一支用 clockadj_max_freq() */
        clockadj_init(c->clkid);
    } else {                               /* ④ 什么都没有：系统时钟 */
        c->clkid = CLOCK_REALTIME;
        c->utc_timescale = 1;
        clockadj_init(c->clkid);
        max_adj = sysclk_max_freq();            /* ← 这一支用 sysclk_max_freq() */
        sysclk_set_leap(0);
    }

    /* 读出"当前已经施加在这个钟上的调整量"；CLOCK_INVALID 时保持 0 */
    if (c->clkid != CLOCK_INVALID)
        fadj = clockadj_get_freq(c->clkid);      /* clock.c:1333-1334 */

    /* 注意第3个参数是 -fadj（取负） */
    c->servo = servo_create(c->config, servo, -fadj, max_adj, sw_ts);  /* :1341 */

    /* ……此后才是 tsproc、端口、UDS 等的创建（clock.c:1346-1444）…… */
}
```

### 时钟调整流程

```c
/* clock.c:2007-2107，改写：保留主干，省略 free_running 旁路、统计与事件通知 */

enum servo_state clock_synchronize(struct clock *c, tmv_t ingress, tmv_t origin)
{
    enum servo_state state = SERVO_UNLOCKED;     /* clock.c:2009 */
    double adj, weight;
    int64_t offset;

    /* 刚跳过钟之后的若干拍，直接返回上次的状态，不做任何调整 */
    if (c->step_window_counter) {                /* clock.c:2013-2019 */
        c->step_window_counter--;
        return c->servo_state;
    }

    c->ingress_ts = ingress;

    /* offset 不是 tmv_sub(ingress, origin)，而是交给 tsproc 用延迟模型算出来的；
       顺带产出这一拍的 weight */
    tsproc_down_ts(c->tsproc, origin, ingress);                     /* :2023 */
    if (tsproc_update_offset(c->tsproc, &c->master_offset, &weight)) {  /* :2025 */
        return state;            /* 延迟还没测出来，这一拍无事可做 */
    }

    if (clock_utc_correct(c, ingress))    /* UTC/闰秒当拍不动钟 */
        return c->servo_state;

    offset = tmv_to_nanoseconds(c->master_offset);                  /* :2045 */
    adj = servo_sample(c->servo, offset, tmv_to_nanoseconds(ingress),
                       weight, &state);                             /* :2046-2047 */
    c->servo_state = state;

    tsproc_set_clock_rate_ratio(c->tsproc, clock_rate_ratio(c));     /* :2050 */

    switch (state) {                                                /* :2052 */
    case SERVO_UNLOCKED:
        break;                               /* 伺服还没准备好，什么都不做 */
    case SERVO_JUMP:
        if (clock_synchronize_locked(c, adj))     /* ①先把频偏归零 :2056 */
            goto servo_unlock;
        if (clockadj_step(c->clkid, -offset))     /* ②再跳相位 :2059 */
            goto servo_unlock;
        c->ingress_ts = tmv_zero();
        tsproc_reset(c->tsproc, 0);               /* :2068 */
        clock_step_window(c);                     /* :2069 起跳过计数 */
        break;
    case SERVO_LOCKED:
        if (clock_synchronize_locked(c, adj))     /* 只调频 :2071-2074 */
            goto servo_unlock;
        break;
    case SERVO_LOCKED_STABLE:
        if (c->write_phase_mode) {                /* 写相位模式 :2076-2081 */
            if (clockadj_set_phase(c->clkid, -offset))
                goto servo_unlock;
            adj = 0;
        } else {                                  /* 否则照样只调频 :2082-2086 */
            if (clock_synchronize_locked(c, adj))
                goto servo_unlock;
        }
        break;
    }

    /* ……此处源码还有 stats 更新、pr_info("master offset …") 与
       clock_notify_event(NOTIFY_TIME_SYNC)（clock.c:2090-2099）…… */

    return state;

servo_unlock:                                     /* clock.c:2103-2106 */
    servo_reset(c->servo);        /* 任何一次 clockadj_* 失败都落到这里 */
    c->servo_state = SERVO_UNLOCKED;
    return SERVO_UNLOCKED;
}
```

上面那个 `clock_synchronize_locked()` 是个三行的包装函数，**所有取负都发生在它里面**，值得单独贴一次：

```c
/* clock.c:1990-2005，完整源码 */

static int clock_synchronize_locked(struct clock *c, double adj)
{
    if (c->sanity_check) {
        clockcheck_freq(c->sanity_check, clockadj_get_freq(c->clkid));
    }
    if (clockadj_set_freq(c->clkid, -adj)) {     /* ← 取负在这里 */
        return -1;
    }
    if (c->clkid == CLOCK_REALTIME) {
        sysclk_set_sync();
    }
    if (c->sanity_check) {
        clockcheck_set_freq(c->sanity_check, -adj);
    }
    return 0;
}
```

## 小结：PHC操作的核心要点

**用户态到内核态的桥梁**：
- 文件描述符 → clockid_t
- 标准时钟API操作PHC

**关键函数**：
- phc_open/close：打开/关闭设备
- clockadj_set_freq：设置频率（系统钟还顺带动 tick）
- clockadj_set_phase：设置相位（只在写相位模式下用）
- clockadj_step：时钟跳变
- clockadj_init / clockadj_max_freq：tick 机制的来源与最大调频能力
- sysclk_set_leap / sysclk_set_tai_offset / sysclk_set_sync：系统钟专属的闰秒、TAI、STA_UNSYNC 处理

**系统调用**：
- clock_adjtime：核心系统调用
- struct timex：丰富的控制参数

**三条调整路径**（与 clock.c:2052-2088 的分支一一对应）：
- 调频 `clockadj_set_freq()`：`SERVO_LOCKED`，以及非写相位模式下的 `SERVO_LOCKED_STABLE` 常态路径
- 调相位 `clockadj_set_phase()`：只在 `write_phase_mode` 且 `SERVO_LOCKED_STABLE`（`clock.c:2076-2079`）
- 立即跳变 `clockadj_step()`：只在 `SERVO_JUMP`，并且**先 set_freq 再 step**（`clock.c:2056` → `:2059`）
