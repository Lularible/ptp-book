# 4.5 Compiling, Running, and Testing

Source version note: this chapter is based on ptp-lite v1.0.0 (2026-04-10).

## The Makefile

```makefile
# ==================== compiler configuration ====================
CC = gcc
CC_ARM64 = aarch64-linux-gnu-gcc
CC_ARM32 = arm-linux-gnueabihf-gcc

CFLAGS = -Wall -Wextra -O2 -std=gnu11
LDFLAGS = -lrt -lm
CFLAGS_ARM64 = -Wall -Wextra -O2 -std=gnu11
LDFLAGS_ARM64 = -lrt -lm
CFLAGS_ARM32 = -Wall -Wextra -O2 -std=gnu11
LDFLAGS_ARM32 = -lrt -lm

all: ptp_master ptp_slave

ptp_master: ptp_master.c ptp_message.c ptp_servo.c
	$(CC) $(CFLAGS) -o $@ $^ $(LDFLAGS)

ptp_slave: ptp_slave.c ptp_message.c ptp_servo.c
	$(CC) $(CFLAGS) -o $@ $^ $(LDFLAGS)

arm64: ptp_master_arm64 ptp_slave_arm64
ptp_master_arm64: ptp_master.c ptp_message.c ptp_servo.c
	$(CC_ARM64) $(CFLAGS_ARM64) -o $@ $^ $(LDFLAGS_ARM64)
ptp_slave_arm64: ptp_slave.c ptp_message.c ptp_servo.c
	$(CC_ARM64) $(CFLAGS_ARM64) -o $@ $^ $(LDFLAGS_ARM64)

arm32: ptp_master_arm32 ptp_slave_arm32
ptp_master_arm32: ptp_master.c ptp_message.c ptp_servo.c
	$(CC_ARM32) $(CFLAGS_ARM32) -o $@ $^ $(LDFLAGS_ARM32)
ptp_slave_arm32: ptp_slave.c ptp_message.c ptp_servo.c
	$(CC_ARM32) $(CFLAGS_ARM32) -o $@ $^ $(LDFLAGS_ARM32)

clean:
	rm -f ptp_master ptp_slave *.o
clean-arm64:
	rm -f ptp_master_arm64 ptp_slave_arm64 *.o
clean-arm32:
	rm -f ptp_master_arm32 ptp_slave_arm32 *.o
clean-all: clean clean-arm64 clean-arm32

all-arch: all arm64 arm32
help:
	@echo "PTP-lite targets: make / arm64 / arm32 / all-arch / clean*"
```

## Build

```bash
make                 # x86
make arm64           # aarch64
make arm32           # armhf
make all-arch        # all three
```

Artifacts: `ptp_master`, `ptp_master_arm64`, `ptp_master_arm32`, and the matching slaves.

## Cross-compile toolchains

```bash
# Debian/Ubuntu
sudo apt install gcc-aarch64-linux-gnu gcc-arm-linux-gnueabihf

# Fedora/CentOS
sudo yum install gcc-aarch64-linux-gnu gcc-arm-linux-gnueabihf
```

## Test (two machines)

```bash
# 1. build
make

# 2. machine A: master
sudo ./ptp_master eth0

# 3. machine B: slave, ideally on a direct link
sudo ./ptp_slave eth0

# 4. watch the slave output
Received Sync seq=0
Follow_Up seq=0: t1=... t2=...
Delay_Resp: delay=2000 ns offset=-5000 ns
CLOCK JUMP: -5000 ns (new time: ...)
...
```

The slave should converge to small offsets and stable frequency.

## Verify

```bash
file ptp_master       # ELF 64-bit x86-64
file ptp_master_arm64 # ELF 64-bit ARM aarch64
file ptp_master_arm32 # ELF 32-bit ARM
```

On the slave, compare `date` on both machines; on a tuned network, expect sub-millisecond or better (software timestamps cap the floor).

## Summary

- One Makefile, three architectures.
- Run master on one host, slave on another; watch the offsets collapse.
- Verification = check the servo rows + `date` on both ends.

Firewall, multicast, packets dropped, and the standard set of levers for precision.
